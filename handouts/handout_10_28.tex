\begin{exe}

    k'i'y t’imḵ’aax g̱an sdo'o=hl t’imḵ’aax win hitxw=hl gat-m g̱an \\
    one armspan \Pcnj{} half=\Cn{} armspan \Comp{} stand=\Cn{} man-\Attr{} wood \\
    \glt `one and a half armspans (9 feet) is the height of the totem pole'
\end{exe}

\begin{exe}
    \ex \glll
    K’i’yhl t’imḵ’aax g̱a-’nekwhl *(ha’niitx̱ooḵxw) ii *(k’i’yhl) sdo’ohl ?(t’imḵ’aax g̱a-le’y=hl) ha’niitx̱ooḵxw. \\
    k’i’y=hl t’imḵ’aax g̱a-’nekw=hl *(ha’niitx̱ooḵxw) ii *(k’i’y=hl) sdo’o=hl ?(t’imḵ’aax g̱a-le’y=hl) ha’niitx̱ooḵxw. \\
    one=\Cn{} armspan \Dist-long=\Cn{} \Ins-on-eat-\Val{} one half=\Cn{} armspan \Dist-wide=\Cn{} \Ins-on-eat-\Val{} \\
    \glt `This table is five feet long and three feet wide.'
\end{exe}

\begin{exe}
    \ex \glll (g̱ay k’aa) ‘wii ’nekw=hl *(ha’niitx̱ooḵxw) ahl (\textbf{win} \textbf{sdo’ohl}) g̱ale’yhlit. \\
    (g̱ay k’aa) ‘wii ’nekw=hl *(ha’niitx̱ooḵxw) a[-t]=hl (\textbf{win} \textbf{sdo’o=hl}) g̱a-le’y=hl-t \\
    \Cntr{} exceedingly big long=\Cn{} table \Prep[-3.\Ii]=\Cn{} \Comp{} half=\Cn{} \Dist-long=\Cn{}-3.\Ii \\ 
    \glt `The table is longer than it is wide (\textbf{by half}).'
\end{exe}
